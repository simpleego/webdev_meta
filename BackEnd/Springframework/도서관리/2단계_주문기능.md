# **주문(Order)** 기능

---

## 📦 주문(Order) 기능 구현 절차

> DB 테이블 → DTO → Repository → Service → Controller → JSP 화면

---

### 1️⃣ DB 테이블 생성 (MySQL 기준)

```sql
CREATE TABLE orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT NOT NULL,
    book_id INT NOT NULL,
    quantity INT DEFAULT 1,
    status VARCHAR(20) DEFAULT 'PENDING',
    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

> 주문 상태는 `PENDING`, `COMPLETED`, `CANCELLED` 등으로 관리할 수 있습니다.

### 3️⃣ Repository (OrderRepository.java)

```java
package com.example.bookstore.order;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.core.RowMapper;
import org.springframework.stereotype.Repository;

import java.util.List;

@Repository
public class OrderRepository {

    private final JdbcTemplate jdbc;

    public OrderRepository(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    private RowMapper<Order> mapper = (rs, rowNum) -> {
        Order o = new Order();
        o.setId(rs.getInt("id"));
        o.setCustomerId(rs.getInt("customer_id"));
        o.setBookId(rs.getInt("book_id"));
        o.setQuantity(rs.getInt("quantity"));
        o.setStatus(rs.getString("status"));
        o.setOrderDate(rs.getString("order_date"));
        return o;
    };

    public void create(Order o) {
        String sql = "INSERT INTO orders (customer_id, book_id, quantity, status) VALUES (?, ?, ?, ?)";
        jdbc.update(sql, o.getCustomerId(), o.getBookId(), o.getQuantity(), o.getStatus());
    }

    public List<Order> findByCustomer(int customerId) {
        String sql = "SELECT * FROM orders WHERE customer_id = ? ORDER BY order_date DESC";
        return jdbc.query(sql, mapper, customerId);
    }

    public void cancel(int id) {
        String sql = "UPDATE orders SET status = 'CANCELLED' WHERE id = ?";
        jdbc.update(sql, id);
    }
}
```

### 4️⃣ Service (OrderService.java)

```java
package com.example.bookstore.order;

import org.springframework.stereotype.Service;

import java.util.List;

@Service
public class OrderService {

    private final OrderRepository repo;

    public OrderService(OrderRepository repo) {
        this.repo = repo;
    }

    public void placeOrder(Order o) {
        o.setStatus("PENDING");
        repo.create(o);
    }

    public List<Order> getOrdersByCustomer(int customerId) {
        return repo.findByCustomer(customerId);
    }

    public void cancelOrder(int id) {
        repo.cancel(id);
    }
}
```

### 5️⃣ Controller (OrderController.java)

```java
package com.example.bookstore.order;

import jakarta.servlet.http.HttpSession;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@Controller
@RequestMapping("/order")
public class OrderController {

    private final OrderService service;

    public OrderController(OrderService service) {
        this.service = service;
    }

    @GetMapping("/list")
    public String list(HttpSession session, Model model) {
        String username = (String) session.getAttribute("username");
        if (username == null) return "redirect:/customer/login";

        int customerId = getCustomerIdByUsername(username); // 구현 필요
        List<Order> orders = service.getOrdersByCustomer(customerId);
        model.addAttribute("orders", orders);
        return "order/list";
    }

    @PostMapping("/create")
    public String create(@RequestParam int bookId,
                         @RequestParam int quantity,
                         HttpSession session) {
        String username = (String) session.getAttribute("username");
        if (username == null) return "redirect:/customer/login";

        int customerId = getCustomerIdByUsername(username); // 구현 필요

        Order o = new Order();
        o.setCustomerId(customerId);
        o.setBookId(bookId);
        o.setQuantity(quantity);
        service.placeOrder(o);

        return "redirect:/order/list";
    }

    @PostMapping("/cancel")
    public String cancel(@RequestParam int orderId) {
        service.cancelOrder(orderId);
        return "redirect:/order/list";
    }

    // 임시 메서드: 실제로는 CustomerService에서 가져와야 함
    private int getCustomerIdByUsername(String username) {
        return 1; // 예시용
    }
}
```
